def current_user():
    user_id = session.get("user_id")
    return db.session.get(User, user_id) if user_id else None


def login_required(view):
    @wraps(view)
    def wrapped(*args, **kwargs):
        if not current_user():
            flash("Please log in to continue.", "warning")
            return redirect(url_for("login", next=request.path))
        return view(*args, **kwargs)
    return wrapped


def admin_required(view):
    @wraps(view)
    def wrapped(*args, **kwargs):
        user = current_user()
        if not user or user.role != "admin":
            flash("Admin access is required.", "warning")
            return redirect(url_for("admin_login"))
        return view(*args, **kwargs)
    return wrapped


def stop_position(route_id, stop_id):
    row = db.session.execute(
        select(RouteStop.position).where(
            RouteStop.route_id == route_id, RouteStop.stop_id == stop_id
        )
    ).scalar_one_or_none()
    return row


def route_matches(bus, source_id, destination_id):
    source_pos = stop_position(bus.route_id, source_id)
    dest_pos = stop_position(bus.route_id, destination_id)
    if source_pos is None or dest_pos is None or source_pos >= dest_pos:
        return None
    ordered = bus.route.stops
    return {
        "source_pos": source_pos,
        "dest_pos": dest_pos,
        "segment_count": dest_pos - source_pos,
        "intermediate": [item.stop for item in ordered if source_pos < item.position < dest_pos],
        "all_stops": [item.stop for item in ordered],
    }


def matching_buses(source_id, destination_id):
    results = []
    for bus in Bus.query.filter_by(active=True).all():
        match = route_matches(bus, source_id, destination_id)
        if match:
            results.append((bus, match))
    return sorted(results, key=lambda item: (item[1]["segment_count"], item[0].first_bus_time))

    return {
        "current_user": current_user(),
        "now_ist": now,
        "default_travel_at": (now + timedelta(minutes=2)).replace(second=0, microsecond=0),
    }


@app.template_filter("ist")
def format_ist(value, fmt="%d %b %Y, %I:%M %p"):
    if not value:
        return "—"
    if value.tzinfo is None:
        value = value.replace(tzinfo=IST)
    return value.astimezone(IST).strftime(fmt)

@app.get("/bus-stops")
def bus_stops():
    query = request.args.get("q", "").strip()
    q = BusStop.query
    if query:
        q = q.filter(BusStop.name.ilike(f"%{query}%") | BusStop.area.ilike(f"%{query}%"))
    stops = q.order_by(BusStop.name).all()
    stop_routes = {}
    for stop in stops:
        stop_routes[stop.id] = [
            rs.route for rs in RouteStop.query.filter_by(stop_id=stop.id).all()
        ]
    return render_template("bus_stops.html", stops=stops, stop_routes=stop_routes, query=query)


@app.get("/nearest-stop")
def nearest_stop():
    lat = request.args.get("lat", type=float)
    lon = request.args.get("lon", type=float)
    nearest = None
    if lat is not None and lon is not None:
        ranked = []
        for stop in BusStop.query.filter(BusStop.latitude.isnot(None), BusStop.longitude.isnot(None)).all():
            dlat = math.radians(stop.latitude - lat)
            dlon = math.radians(stop.longitude - lon)
            a = math.sin(dlat / 2) ** 2 + math.cos(math.radians(lat)) * math.cos(math.radians(stop.latitude)) * math.sin(dlon / 2) ** 2
            distance = 6371 * 2 * math.atan2(math.sqrt(a), math.sqrt(1 - a))
            ranked.append((distance, stop))
        if ranked:
            nearest = min(ranked, key=lambda item: item[0])
    return render_template("nearest_stop.html", nearest=nearest)


@app.get("/schedule")
def schedule():
    term = request.args.get("q", "").strip()
    query = Bus.query.join(BusRoute)
    if term:
        query = query.filter(
            Bus.bus_number.ilike(f"%{term}%") |
            Bus.name.ilike(f"%{term}%") |
            BusRoute.name.ilike(f"%{term}%") |
            BusRoute.id.in_(
                db.session.query(RouteStop.route_id).join(BusStop).filter(BusStop.name.ilike(f"%{term}%"))
            )
        )
    return render_template("schedule.html", buses=query.filter(Bus.active.is_(True)).all(), query=term)


@app.route("/booking/<int:bus_id>", methods=["GET", "POST"])
@login_required
def booking(bus_id):
    bus = db.get_or_404(Bus, bus_id)
    source_id = request.values.get("source", type=int)
    destination_id = request.values.get("destination", type=int)
    match = route_matches(bus, source_id, destination_id) if source_id and destination_id else None
    if not match:
        flash("That bus does not serve the selected stops in that direction.", "danger")
        return redirect(url_for("find_bus"))
    categories = {"Adult": 1.0, "Child": 0.5, "Senior Citizen": 0.7}
    if request.method == "POST":
        category = request.form.get("category", "")
        count = request.form.get("passenger_count", type=int)
        travel_text = request.form.get("travel_at", "")
        try:
            travel_at = datetime.strptime(travel_text, "%Y-%m-%dT%H:%M").replace(tzinfo=IST)
        except ValueError:
            travel_at = None
        if category not in categories or not count or not 1 <= count <= 8:
            flash("Select a passenger category and enter 1–8 passengers.", "danger")
        elif not travel_at or travel_at < datetime.now(IST):
            flash("Choose a valid travel date and time that is not in the past.", "danger")
        else:
            fare_each = round(float(bus.base_fare_per_segment) * match["segment_count"] * categories[category], 2)
            item = Booking(
                user_id=current_user().id, bus_id=bus.id,
                source_stop_id=source_id, destination_stop_id=destination_id,
                travel_at=travel_at, passenger_category=category,
                passenger_count=count, fare_each=fare_each,
                total_fare=round(fare_each * count, 2),
            )
            db.session.add(item)
            db.session.commit()
            return redirect(url_for("payment", booking_id=item.id))
    return render_template(
        "booking.html", bus=bus, match=match,
        source=db.get_or_404(BusStop, source_id),
        destination=db.get_or_404(BusStop, destination_id),
    )


@app.route("/payment/<int:booking_id>", methods=["GET", "POST"])
@login_required
def payment(booking_id):
    item = db.get_or_404(Booking, booking_id)
    if item.user_id != current_user().id:
        abort(403)
    if item.status == "paid":
        return redirect(url_for("ticket", reference=item.reference))
    if request.method == "POST":
        mode = request.form.get("mode")
        if mode not in ("UPI", "Card", "Net Banking"):
            flash("Choose a payment mode.", "danger")
        else:
            item.status = "paid"
            db.session.add(Payment(booking=item, mode=mode, amount=item.total_fare))
            db.session.commit()
            return redirect(url_for("ticket", reference=item.reference))
    return render_template("payment.html", booking=item)

@app.route("/admin/routes", methods=["GET", "POST"])
@admin_required
def admin_routes():
    if request.method == "POST":
        name = request.form.get("name", "").strip()
        description = request.form.get("description", "").strip()
        stop_names = [part.strip() for part in request.form.get("stops", "").split(",") if part.strip()]
        stops = [BusStop.query.filter(func.lower(BusStop.name) == s.lower()).first() for s in stop_names]
        if not name or BusRoute.query.filter_by(name=name).first():
            flash("Enter a route name that is not already used.", "danger")
        elif len(stops) < 2 or any(stop is None for stop in stops) or len(set(s.id for s in stops if s)) != len(stops):
            flash("Enter at least two existing stop names in travel order, separated by commas.", "danger")
        else:
            route = BusRoute(name=name, description=description)
            db.session.add(route)
            db.session.flush()
            for position, stop in enumerate(stops):
                db.session.add(RouteStop(route_id=route.id, stop_id=stop.id, position=position))
            db.session.commit()
            flash("Route added with its ordered stops.", "success")
            return redirect(url_for("admin_routes"))
    return render_template("admin_routes.html", routes=BusRoute.query.order_by(BusRoute.name).all(), stops=BusStop.query.order_by(BusStop.name).all())

-- PostgreSQL schema for Bus Route Finder and Ticket Booking System.
-- The app's `flask --app app init-db` command creates these tables and seeds
-- fictional demonstration records. This file is also provided for inspection
-- and manual database setup.
SET TIME ZONE 'Asia/Kolkata';

CREATE TABLE IF NOT EXISTS "user" (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(120) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    mobile VARCHAR(20) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(20) NOT NULL DEFAULT 'user' CHECK (role IN ('user', 'admin')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS bus_stop (
    id SERIAL PRIMARY KEY,
    name VARCHAR(120) NOT NULL UNIQUE,
    area VARCHAR(120) NOT NULL DEFAULT '',
    latitude DOUBLE PRECISION,
    longitude DOUBLE PRECISION,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS bus_route (
    id SERIAL PRIMARY KEY,
    name VARCHAR(120) NOT NULL UNIQUE,
    description TEXT NOT NULL DEFAULT '',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS route_stop (
    id SERIAL PRIMARY KEY,
    route_id INTEGER NOT NULL REFERENCES bus_route(id) ON DELETE CASCADE,
    stop_id INTEGER NOT NULL REFERENCES bus_stop(id) ON DELETE CASCADE,
    position INTEGER NOT NULL CHECK (position >= 0),
    UNIQUE (route_id, position),
    UNIQUE (route_id, stop_id)
);

CREATE TABLE IF NOT EXISTS bus (
    id SERIAL PRIMARY KEY,
    bus_number VARCHAR(30) NOT NULL UNIQUE,
    name VARCHAR(120) NOT NULL,
    route_id INTEGER NOT NULL REFERENCES bus_route(id),
    first_bus_time VARCHAR(5) NOT NULL DEFAULT '06:00',
    last_bus_time VARCHAR(5) NOT NULL DEFAULT '22:00',
    frequency_mins INTEGER NOT NULL DEFAULT 20 CHECK (frequency_mins > 0),
    base_fare_per_segment NUMERIC(8,2) NOT NULL DEFAULT 10 CHECK (base_fare_per_segment > 0),
    active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS booking (
    id SERIAL PRIMARY KEY,
    reference VARCHAR(24) NOT NULL UNIQUE,
    user_id INTEGER NOT NULL REFERENCES "user"(id),
    bus_id INTEGER NOT NULL REFERENCES bus(id),
    source_stop_id INTEGER NOT NULL REFERENCES bus_stop(id),
    destination_stop_id INTEGER NOT NULL REFERENCES bus_stop(id),
    travel_at TIMESTAMPTZ NOT NULL,
    passenger_category VARCHAR(30) NOT NULL
        CHECK (passenger_category IN ('Adult', 'Child', 'Senior Citizen')),
    passenger_count INTEGER NOT NULL CHECK (passenger_count BETWEEN 1 AND 8),
    fare_each NUMERIC(8,2) NOT NULL CHECK (fare_each >= 0),
    total_fare NUMERIC(8,2) NOT NULL CHECK (total_fare >= 0),
    status VARCHAR(20) NOT NULL DEFAULT 'pending'
        CHECK (status IN ('pending', 'paid')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CHECK (source_stop_id <> destination_stop_id)
);

CREATE TABLE IF NOT EXISTS payment (
    id SERIAL PRIMARY KEY,
    payment_id VARCHAR(32) NOT NULL UNIQUE,
    booking_id INTEGER NOT NULL UNIQUE REFERENCES booking(id),
    mode VARCHAR(30) NOT NULL CHECK (mode IN ('UPI', 'Card', 'Net Banking')),
    amount NUMERIC(8,2) NOT NULL CHECK (amount >= 0),
    status VARCHAR(20) NOT NULL DEFAULT 'successful'
        CHECK (status IN ('successful')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS complaint (
    id SERIAL PRIMARY KEY,
    reference VARCHAR(24) NOT NULL UNIQUE,
    user_id INTEGER REFERENCES "user"(id),
    name VARCHAR(120) NOT NULL,
    contact VARCHAR(255) NOT NULL,
    bus_number VARCHAR(30) NOT NULL DEFAULT '',
    route_name VARCHAR(120) NOT NULL DEFAULT '',
    description TEXT NOT NULL,
    status VARCHAR(30) NOT NULL DEFAULT 'submitted'
        CHECK (status IN ('submitted', 'in progress', 'resolved')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS ix_user_email ON "user"(email);
CREATE INDEX IF NOT EXISTS ix_booking_user_created ON booking(user_id, created_at DESC);
CREATE INDEX IF NOT EXISTS ix_complaint_created ON complaint(created_at DESC);

# Bus Route Finder and Ticket Booking System

A beginner-friendly college demonstration app built with Python, Flask, PostgreSQL, HTML, CSS, and JavaScript. It includes route search, schedules, user accounts, automatic fare calculation, demo ticket booking/payment, complaints, nearest-stop lookup, and basic admin management.

**No live bus tracking and no real payment processing are included.** All bus data is fictional sample data. A printed ticket is not valid for travel.

## Features

- Search buses between two stops; direct routes and fewer stops appear first.
- Browse stops, route stops, and sample bus schedules.
- Find the closest sample stop using optional browser location permission, or search manually.
- Register and log in with securely hashed passwords.
- Book one passenger category at a time (Adult, Child, Senior Citizen), up to 8 passengers.
- Fare is computed from route segments, the bus's database-stored per-segment fare, passenger category, and passenger count.
- Complete a simulated UPI, card, or net-banking payment without entering payment credentials.
- View and print a ticket and browse previous bookings.
- Submit complaints with automatically generated reference and database timestamp.
- Admin can add/remove buses and stops, create ordered routes, and update complaint status.
- Database timestamps are generated by PostgreSQL and displayed in Asia/Kolkata time.

## 1. Install on Ubuntu/Linux

Check Python:

```bash
python3 --version
```

Install Python, virtual-environment tools, and PostgreSQL:

```bash
sudo apt update
sudo apt install python3 python3-venv python3-pip postgresql postgresql-contrib
sudo systemctl enable --now postgresql
sudo systemctl status postgresql
```

Create a PostgreSQL account and database. Replace the sample password with a strong password; use the same value in `DATABASE_URL` in your `.env` file.

```bash
sudo -u postgres psql
```

At the `postgres=#` prompt:

```sql
CREATE USER bus_app WITH PASSWORD 'choose-a-strong-password';
CREATE DATABASE bus_finder OWNER bus_app;
\q
```

The first command opens PostgreSQL's command line as its administrator. `CREATE USER` makes the app's database login, and `CREATE DATABASE` creates the app's data store with that user as owner.

## 2. Configure and install the app

From this folder:

```bash
cp .env.example .env
python3 -m venv venv
source venv/bin/activate
pip install -r requirements.txt
```

Edit `.env` and set:

- `DATABASE_URL` — your PostgreSQL username, password, host, port, and database.
- `FLASK_SECRET_KEY` — a long, private random value used to protect sessions.
- `ADMIN_EMAIL` and `ADMIN_PASSWORD` — the initial admin account. Choose a strong password before running the initializer.
- `FLASK_DEBUG=1` only for local development. Use `0` outside development.

The values shown in `.env.example` are placeholders. Do not publish a real `.env` file.

## 3. Create tables and sample records

```bash
flask --app app init-db
```

This creates the schema and inserts fictional sample stops, routes, and buses. It also creates the initial admin account from `ADMIN_EMAIL` and `ADMIN_PASSWORD`. Run this once after configuring the database. To inspect the corresponding PostgreSQL schema, see `database.sql`.

## 4. Start the website

```bash
flask --app app run
```

Open <http://127.0.0.1:5000> in Firefox or Chrome. Stop the development server with **Ctrl+C** in the terminal.

### Initial admin sign-in

Open <http://127.0.0.1:5000/admin/login> and use the email/password set in `.env`. The default values in `.env.example` are not safe for deployment; change them before initializing the database.

## Fare rule

The app counts the number of adjacent stop-to-stop segments between the selected source and destination:

```text
adult fare per passenger = bus base fare per segment × number of segments
child fare               = adult fare × 0.50
senior citizen fare      = adult fare × 0.70
total                    = fare per passenger × passenger count
```

The browser previews the fare for convenience, but the Flask backend recalculates it from database records when a booking is submitted. The payment page is a simulation and stores a demo payment record only after the user confirms a payment mode.

## Add a route in the admin panel

1. Add any missing stops under **Admin → Manage stops**.
2. Under **Manage routes**, create a route and enter the existing stop names in travel order, separated by commas. Names must match the stop list.
3. Under **Manage buses**, assign a bus number, bus name, route, sample schedule, and adult fare per segment.

The seed data contains three sample routes and buses. Admin route stop order controls which direction a bus can be booked.

## Database timestamps and timezone

`created_at` and `updated_at` fields use PostgreSQL `TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP`. The app configures its PostgreSQL session to `Asia/Kolkata`, validates travel date/time on the server in IST, and formats booking/complaint/payment timestamps in IST. Travel time is chosen by the user; booking, complaint, and payment creation times are generated automatically by PostgreSQL.

## Quick manual test

1. Create a regular account at `/register` and log in.
2. Search `Central Station` → `University Gate`; compare the matching route.
3. Select a bus, book an adult ticket for a future time, then complete the demo payment.
4. Print the ticket and check **My tickets**.
5. Submit a complaint; sign in as admin to change its status.
6. Try a child/senior fare, a route in the reverse direction, and a same-stop search.

## Troubleshooting

- **`psql` cannot connect:** run `sudo systemctl status postgresql`; start it with `sudo systemctl start postgresql`.
- **Database authentication failed:** check the username/password in `.env`, and ensure the role owns `bus_finder`.
- **`ModuleNotFoundError`:** activate the virtual environment with `source venv/bin/activate`, then run `pip install -r requirements.txt`.
- **Address already in use:** run `flask --app app run --port 5001` and open `http://127.0.0.1:5001`.
- **No sample routes appear:** run `flask --app app init-db` after configuring `DATABASE_URL`.
- **Nearest stop asks for location:** allow permission in the browser, or use the manual stop search. Browser location generally requires localhost or HTTPS.

## Academic scope and limitations

This version is designed as a straightforward starting point for a TY BCA practical/viva. It uses sample data, a simulated payment, and basic admin CRUD. It does not integrate a transport authority, real payment gateway, live GPS, ticket QR validation, email/SMS delivery, or production-grade deployment monitoring.
