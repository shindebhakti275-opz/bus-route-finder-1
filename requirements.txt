@app.template_filter("ist")
def format_ist(
    value,
    format_string="%d %b %Y"
):

    if not value:
        return ""

    try:

        if value.tzinfo is None:
            value = value.replace(
                tzinfo=timezone.utc
            )

        value = value.astimezone(IST)

        return value.strftime(
            format_string
        )

    except Exception:

        return str(value)


@app.template_filter("currency")
def currency_filter(value):

    try:
        return f"₹{float(value):.2f}"

    except Exception:
        return "₹0.00"

def csrf_token():

    token = session.get(
        "_csrf_token"
    )

    if not token:

        token = os.urandom(24).hex()

        session["_csrf_token"] = token

    return token

def login_required(function):

    @wraps(function)
    def wrapper(*args, **kwargs):

        if not session.get(
            "user_id"
        ):

            flash(
                "Please login first.",
                "warning"
            )

            return redirect(
                url_for("login")
            )

        return function(
            *args,
            **kwargs
        )

    return wrapper


def admin_required(function):

    @wraps(function)
    def wrapper(*args, **kwargs):

        user = get_current_user()

        if not user:

            flash(
                "Please login first.",
                "warning"
            )

            return redirect(
                url_for("login")
            )

        if (
            not user.is_admin
            and user.role != "admin"
        ):

            flash(
                "Admin access required.",
                "danger"
            )

            return redirect(
                url_for("index")
            )

        return function(
            *args,
            **kwargs
        )

    return wrapper

# ============================================================
# DYNAMIC ROUTE SEARCH
# ============================================================

def build_route_graph():

    graph = {}

    routes = BusRoute.query.all()

    for route in routes:

        stop_ids = get_route_stop_ids(
            route.id
        )

        buses = get_route_buses(
            route.id
        )

        if not stop_ids or not buses:
            continue

        for index, stop_id in enumerate(
            stop_ids
        ):

            graph.setdefault(
                stop_id,
                []
            )

            for next_index in range(
                index + 1,
                len(stop_ids)
            ):

                destination_id = (
                    stop_ids[next_index]
                )

                graph[stop_id].append(
                    {
                        "route": route,
                        "bus": buses[0],
                        "destination_id": destination_id,
                        "stops": stop_ids[
                            index:next_index + 1
                        ]
                    }
                )

    return graph


def find_dynamic_route(
    source_id,
    destination_id
):

    if source_id == destination_id:
        return None

    graph = build_route_graph()

    queue = deque()

    queue.append(
        (
            source_id,
            []
        )
    )

    visited = {
        source_id
    }

    while queue:

        current_stop, journey = (
            queue.popleft()
        )

        for connection in graph.get(
            current_stop,
            []
        ):

            next_stop = connection[
                "destination_id"
            ]

            new_journey = (
                journey
                + [connection]
            )

            if next_stop == destination_id:

                return new_journey

            if next_stop not in visited:

                visited.add(
                    next_stop
                )

                if len(new_journey) < 5:

                    queue.append(
                        (
                            next_stop,
                            new_journey
                        )
                    )

    return None


def make_journey_details(
    journey
):

    result = []

    total_fare = 0

    for leg in journey:

        bus = leg["bus"]

        stop_ids = leg["stops"]

        fare_per_segment = float(
            bus.base_fare_per_segment
            or 10
        )

        segments = max(
            len(stop_ids) - 1,
            1
        )

        fare = (
            fare_per_segment
            * segments
        )

        total_fare += fare

        stops = [
            db.session.get(
                BusStop,
                stop_id
            )
            for stop_id in stop_ids
        ]

        result.append(
            {
                "bus": bus,
                "route": leg["route"],
                "stops": stops,
                "fare": fare
            }
        )

    return result, total_fare

@app.route(
    "/register",
    methods=["GET", "POST"]
)
def register():

    if request.method == "POST":

        name = (
            request.form.get("name")
            or request.form.get(
                "full_name"
            )
            or ""
        ).strip()

        email = (
            request.form.get("email")
            or ""
        ).strip().lower()

        mobile = (
            request.form.get("mobile")
            or ""
        ).strip()

        password = (
            request.form.get("password")
            or ""
        )

        confirm_password = (
            request.form.get(
                "confirm_password"
            )
            or request.form.get(
                "password_confirmation"
            )
            or ""
        )

        if (
            not name
            or not email
            or not mobile
            or not password
        ):

            flash(
                "Please fill all required fields.",
                "danger"
            )

            return render_template(
                "register.html"
            )

        if (
            confirm_password
            and password != confirm_password
        ):

            flash(
                "Passwords do not match.",
                "danger"
            )

            return render_template(
                "register.html"
            )

        existing = User.query.filter_by(
            email=email
        ).first()

        if existing:

            flash(
                "Email already registered.",
                "danger"
            )

            return render_template(
                "register.html"
            )

        user = User(
            name=name,
            full_name=name,
            email=email,
            mobile=mobile,
            password_hash=generate_password_hash(
                password
            ),
            role="user",
            is_admin=False
        )

        db.session.add(user)
        db.session.commit()

        flash(
            "Registration successful. Please login.",
            "success"
        )

        return redirect(
            url_for("login")
        )

    return render_template(
        "register.html"
    )

@app.route(
    "/login",
    methods=["GET", "POST"]
)

    if request.method == "POST":

        email = (
            request.form.get("email")
            or ""
        ).strip().lower()

        password = (
            request.form.get("password")
            or ""
        )

        user = User.query.filter_by(
            email=email
        ).first()

        if (
            user
            and check_password_hash(
                user.password_hash,
                password
            )
        ):

            session["user_id"] = user.id

            flash(
                "Login successful.",
                "success"
            )

            if (
                user.is_admin
                or user.role == "admin"
            ):

                return redirect(
                    url_for(
                        "admin_dashboard"
                    )
                )

            return redirect(
                url_for("index")
            )

        flash(
            "Invalid email or password.",
            "danger"
        )

    return render_template(
        "login.html"
    )

@app.route(
    "/find-bus",
    methods=["GET", "POST"]
)
def find_bus():

    stops = BusStop.query.order_by(BusStop.name).all()

    options = []
    results = None
    journey = None
    total_fare = 0

    source_id = request.values.get("source", type=int)
    destination_id = request.values.get("destination", type=int)

    if source_id and destination_id:

        if source_id == destination_id:
            flash(
                "Source and destination cannot be the same.",
                "warning"
            )

        else:
            routes = BusRoute.query.order_by(BusRoute.id).all()

            # -------------------------------------------------
            # 1. DIRECT BUS SEARCH
            # -------------------------------------------------

            direct_options = []

            for route in routes:

                route_stops = get_route_stops(route.id)
                buses = get_route_buses(route.id)

                if not route_stops or not buses:
                    continue

                source_position = next(
                    (
                        i for i, stop in enumerate(route_stops)
                        if stop and stop.id == source_id
                    ),
                    None
                )

                destination_position = next(
                    (
                        i for i, stop in enumerate(route_stops)
                        if stop and stop.id == destination_id
                    ),
                    None
                )

                if (
                    source_position is None
                    or destination_position is None
                    or source_position >= destination_position
                ):
                    continue

                journey_stops = route_stops[
                    source_position:destination_position + 1
                ]

                segment_count = max(len(journey_stops) - 1, 1)
                stop_count = len(journey_stops)

                for bus in buses:

                    fare_per_segment = float(
                        bus.base_fare_per_segment or 10
                    )

                    fare = fare_per_segment * segment_count

                    direct_options.append({
                        "type": "direct",
                        "recommended": False,
                        "bus": bus,
                        "match": {
                            "fare": fare,
                            "journey_stops": journey_stops,
                            "segment_count": segment_count,
                            "stop_count": stop_count,
                            "all_stops": route_stops
                        }
                    })

            # If a direct bus exists, show direct buses only.
            if direct_options:

                options = direct_options

            # -------------------------------------------------
            # 2. ONE-TRANSFER / TWO-BUS SEARCH
            # -------------------------------------------------

            else:

                transfer_options = []

                for first_route in routes:

                    first_route_stops = get_route_stops(
                        first_route.id
                    )

                    first_buses = get_route_buses(
                        first_route.id
                    )

                    if not first_route_stops or not first_buses:
                        continue

                    source_position = next(
                        (
                            i for i, stop in enumerate(first_route_stops)
                            if stop and stop.id == source_id
                        ),
                        None
                    )

                    if source_position is None:
                        continue

                    # Every stop after source can be a transfer stop.
                    for first_transfer_position in range(
                        source_position + 1,
                        len(first_route_stops)
                    ):

                        transfer_stop = first_route_stops[
                            first_transfer_position
                        ]

                        if transfer_stop is None:
                            continue

                        transfer_id = transfer_stop.id

                        first_journey_stops = first_route_stops[
                            source_position:first_transfer_position + 1
                        ]

                        first_segment_count = max(
                            len(first_journey_stops) - 1,
                            1
                        )

                        first_stop_count = len(
                            first_journey_stops
                        )

                        for second_route in routes:

                            if second_route.id == first_route.id:
                                continue

                            second_route_stops = get_route_stops(
                                second_route.id
                            )

                            second_buses = get_route_buses(
                                second_route.id
                            )

                            if (
                                not second_route_stops
                                or not second_buses
                            ):
                                continue

                            transfer_position = next(
                                (
                                    i for i, stop in enumerate(second_route_stops)
                                    if stop and stop.id == transfer_id
                                ),
                                None
                            )

                            destination_position = next(
                                (
                                    i for i, stop in enumerate(second_route_stops)
                                    if stop and stop.id == destination_id
                                ),
                                None
                            )

                            if (
                                transfer_position is None
                                or destination_position is None
                                or transfer_position >= destination_position
                            ):
                                continue

                            second_journey_stops = second_route_stops[
                                transfer_position:destination_position + 1
                            ]

                            second_segment_count = max(
                                len(second_journey_stops) - 1,
                                1
                            )

                            second_stop_count = len(
                                second_journey_stops
                            )

                            for first_bus in first_buses:

                                first_fare = (
                                    float(
                                        first_bus.base_fare_per_segment
                                        or 10
                                    )
                                    * first_segment_count
                                )

                                for second_bus in second_buses:

                                    second_fare = (
                                        float(
                                            second_bus.base_fare_per_segment
                                            or 10
                                        )
                                        * second_segment_count
                                    )

                                    transfer_options.append({
                                        "type": "transfer",
                                        "recommended": False,

                                        "bus": first_bus,

                                        "match": {
                                            "fare": first_fare,
                                            "journey_stops":
                                                first_journey_stops,
                                            "segment_count":
                                                first_segment_count,
                                            "stop_count":
                                                first_stop_count,
                                            "all_stops":
                                                first_route_stops
                                        },

                                        "transfer_stop":
                                            transfer_stop,

                                        "second_bus":
                                            second_bus,

                                        "second_match": {
                                            "fare": second_fare,
                                            "journey_stops":
                                                second_journey_stops,
                                            "segment_count":
                                                second_segment_count,
                                            "stop_count":
                                                second_stop_count,
                                            "all_stops":
                                                second_route_stops
                                        },

                                        "total_fare":
                                            first_fare + second_fare,

                                        "total_segments":
                                            first_segment_count
                                            + second_segment_count,

                                        "total_stops":
                                            first_stop_count
                                            + second_stop_count
                                            - 1
                                    })

                # Remove duplicate transfer combinations.
                seen = set()
                unique_transfer_options = []

                for item in transfer_options:

                    key = (
                        item["bus"].id,
                        item["transfer_stop"].id,
                        item["second_bus"].id
                    )

                    if key not in seen:
                        seen.add(key)
                        unique_transfer_options.append(item)

                options = unique_transfer_options

            # -------------------------------------------------
            # FINAL RESULT VALUES
            # -------------------------------------------------

            if options:

                options[0]["recommended"] = True
                results = options

                if options[0]["type"] == "direct":
                    total_fare = options[0]["match"]["fare"]
                else:
                    total_fare = options[0]["total_fare"]

            else:

                flash(
                    "No direct bus or one-transfer route "
                    "was found between these stops.",
                    "warning"
                )

    return render_template(
        "find_bus.html",
        stops=stops,
        options=options,
        results=results,
        journey=journey,
        total_fare=total_fare,
        source_id=source_id,
        destination_id=destination_id
    )

        request.args.get("q", "")
        or ""

    if query:
        search = f"%{query}%"
        stops_query = stops_query.filter(

        )

    stops = (
        stops_query
        .order_by(

        )
        .all()
    )

    stop_routes = {}

    for stop in stops:
        rows = (
            RouteStop.query
            .filter_by(
                stop_id=stop.id
            )
            .order_by(
                RouteStop.position
            )
            .all()
        )

        stop_routes[stop.id] = [
            row.route
            for row in rows
            if row.route is not None
        ]

    return render_template(
        "bus_stops.html",
        stops=stops,
        stop_routes=stop_routes,
        query=query
    )

@app.route(
    "/book",
    methods=["GET", "POST"]
)
@login_required
def book():

    stops = (
        BusStop.query
        .order_by(
            BusStop.name
        )
        .all()
    )

    buses = (
        Bus.query
        .filter_by(
            active=True
        )
        .order_by(
            Bus.id
        )
        .all()
    )

    # Data used by booking.html
    bus_route_data = {}

    for bus in buses:

        bus_route_data[
            str(bus.id)
        ] = {
            "stops": get_route_stop_ids(
                bus.route_id
            ),
            "fare": float(
                bus.base_fare_per_segment
                or 10
            )
        }

    if request.method == "POST":

        bus_id = request.form.get(
            "bus_id",
            type=int
        )

        source_id = request.form.get(
            "source_stop_id",
            type=int
        )

        destination_id = request.form.get(
            "destination_stop_id",
            type=int
        )

        passenger_count = request.form.get(
            "passenger_count",
            type=int
        ) or 1

        fare_type = (
            request.form.get(
                "fare_type"
            )
            or "Full"
        ).strip()

        gender = (
            request.form.get(
                "gender"
            )
            or "Male"
        ).strip()

        # ----------------------------------------------------
        # VALIDATION
        # ----------------------------------------------------

        if (
            passenger_count < 1
            or passenger_count > 8
        ):

            flash(
                "Number of tickets must be between 1 and 8.",
                "danger"
            )

            return redirect(
                url_for("book")
            )

        if fare_type not in (
            "Full",
            "Half"
        ):

            flash(
                "Please select Full or Half ticket.",
                "danger"
            )

            return redirect(
                url_for("book")
            )

        if gender not in (
            "Male",
            "Female"
        ):

            flash(
                "Please select Male or Female.",
                "danger"
            )

            return redirect(
                url_for("book")
            )

        bus = db.session.get(
            Bus,
            bus_id
        )

        source = db.session.get(
            BusStop,
            source_id
        )

        destination = db.session.get(
            BusStop,
            destination_id
        )

        if (
            not bus
            or not source
            or not destination
        ):

            flash(
                "Please select a valid bus, source and destination.",
                "danger"
            )

            return redirect(
                url_for("book")
            )

        if source.id == destination.id:

            flash(
                "Source and destination cannot be the same.",
                "danger"
            )

            return redirect(
                url_for("book")
            )

        stop_ids = get_route_stop_ids(
            bus.route_id
        )

        if (
            source.id not in stop_ids
            or destination.id not in stop_ids
        ):

            flash(
                "This bus does not serve both selected stops.",
                "danger"
            )

            return redirect(
                url_for("book")
            )

        source_position = stop_ids.index(
            source.id
        )

        destination_position = stop_ids.index(
            destination.id
        )

        if (
            source_position
            >= destination_position
        ):

            flash(
                "Please select the stops in the correct journey direction.",
                "danger"
            )

            return redirect(
                url_for("book")
            )

        # ----------------------------------------------------
        # FARE
        # ----------------------------------------------------

        segments = (
            destination_position
            - source_position
        )

        full_fare_each = (
            float(
                bus.base_fare_per_segment
                or 10
            )
            * segments
        )

        if fare_type == "Half":

            fare_each = (
                full_fare_each
                * 0.50
            )

        else:

            fare_each = (
                full_fare_each
            )

        total_fare = (
            fare_each
            * passenger_count
        )

        # ----------------------------------------------------
        # CREATE PENDING BOOKING
        # ----------------------------------------------------

        booking = Booking(

            reference=generate_reference(),

            user_id=session[
                "user_id"
            ],

            bus_id=bus.id,

            source_stop_id=source.id,

            destination_stop_id=destination.id,

            # Automatic current date and time
            travel_at=now_ist(),

            passenger_category="Adult",

            fare_type=fare_type,

            gender=gender,

            passenger_count=passenger_count,

            fare_each=fare_each,

            total_fare=total_fare,

            status="pending"
        )

        db.session.add(
            booking
        )

        db.session.commit()

        # Go to payment
        return redirect(
            url_for(
                "payment",
                reference=booking.reference
            )
        )

    return render_template(
        "booking.html",

        stops=stops,

        buses=buses,

        current_time=now_ist(),

        bus_route_data=bus_route_data
    )

@app.route(
    "/payment/<reference>",
    methods=["GET", "POST"]
)
@login_required
def payment(reference):

    booking = Booking.query.filter_by(
        reference=reference,
        user_id=session[
            "user_id"
        ]
    ).first_or_404()

    existing_payment = Payment.query.filter_by(
        booking_id=booking.id
    ).first()

    if existing_payment:

        return redirect(
            url_for(
                "ticket",
                reference=booking.reference
            )
        )

    if request.method == "POST":

        mode = (
            request.form.get(
                "mode"
            )
            or ""
        ).strip()

        if mode not in (
            "PhonePe",
            "Google Pay"
        ):

            flash(
                "Please select PhonePe or Google Pay.",
                "danger"
            )

            return redirect(
                url_for(
                    "payment",
                    reference=booking.reference
                )
            )

        payment = Payment(

            payment_id=generate_payment_id(),

            booking_id=booking.id,

            mode=mode,

            amount=booking.total_fare,

            status="successful"
        )

        booking.status = "paid"

        db.session.add(
            payment
        )

        db.session.commit()

        flash(
            "Payment completed successfully.",
            "success"
        )

        return redirect(
            url_for(
                "ticket",
                reference=booking.reference
            )
        )

    return render_template(
        "payment.html",
        booking=booking
    )

@app.route(
    "/api/find-route"
)
def api_find_route():

    source_id = request.args.get(
        "source",
        type=int
    )

    destination_id = request.args.get(
        "destination",
        type=int
    )

    if (
        not source_id
        or not destination_id
    ):

        return jsonify(
            {
                "success": False,
                "message": (
                    "Source and destination are required."
                )
            }
        )

    journey = find_dynamic_route(
        source_id,
        destination_id
    )

    if not journey:

        return jsonify(
            {
                "success": False,
                "message": "No route found."
            }
        )

    details, total_fare = (
        make_journey_details(
            journey
        )
    )

    response = []

    for item in details:

        response.append(
            {
                "bus_number": item[
                    "bus"
                ].bus_number,

                "bus_name": item[
                    "bus"
                ].name,

                "route": item[
                    "route"
                ].name,

                "stops": [
                    stop.name
                    for stop in item[
                        "stops"
                    ]
                ],

                "fare": item[
                    "fare"
                ],

                "first_bus": item[
                    "bus"
                ].first_bus_time,

                "last_bus": item[
                    "bus"
                ].last_bus_time,

                "frequency": item[
                    "bus"
                ].frequency_mins
            }
        )

    return jsonify(
        {
            "success": True,
            "journey": response,
            "total_fare": total_fare
        }
    )
